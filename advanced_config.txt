# ===== ADVANCED SETTINGS - rarely changed =====
# These define the trading METHOD itself. The values below are the tested
# defaults; you normally do NOT need to touch them. It is safe to change a
# value, and it is safe to DELETE this whole file (the same defaults still
# apply from the code). Everyday settings live in config.txt.
# After any change here, just re-run:  python scripts/run_scanner.py
# (except the [data] fetch knobs below, which apply on the next Step 1 download.)

[macd]
fast = 12                       # standard MACD fast EMA
slow = 26                       # standard MACD slow EMA
signal = 9                      # standard MACD signal EMA

[divergence]
lookback_days = 60              # recent trading days searched for the pattern (~3 months)
trough_pivot_k = 3              # bars each side that must be >= a bar for it to count as a momentum trough (spec TROUGH_PIVOT_K)
price_window_k = 3              # trading days each side of a dip used to find its true low (and the RSI trough)
tolerance_pct = 0.01            # how close two lows count as "equal" (0.01 = 1%, a double bottom)
price_field = Low               # which price makes the "lower low": Low | Close

[weekly]
ema_set = 11,22,50              # the three weekly averages forming the support band
zone_test = range_overlap       # range_overlap (touch) | close_in_band (stricter: must close inside)
min_weekly_bars_for_zone = 50   # weekly candles a stock needs in total (~1 year) before its 50-week average is used

[rsi]
period = 14                     # RSI length (standard)
smoothing = wilder              # wilder (matches TradingView) | sma
lower_band = 30                 # oversold line: at/above = "uncensored", below = "censored"

[liquidity]
apply_as_filter = true          # turn the liquidity filter on/off
window = 20                     # trading days used to measure average turnover

[ranking]
# How the top list is ordered. Bigger weight = more influence. They need not add to 1.
weight_momentum = 0.30          # strength of the momentum turn
weight_zone_confluence = 0.20   # how tight/centred the weekly support zone is
weight_rsi_quality = 0.15       # healthier (higher) RSI at the low
weight_liquidity = 0.15         # higher daily turnover
weight_recency = 0.10           # fresher signal
weight_volume_expansion = 0.10  # jump in volume on the reversal

[data]
# Download mechanics (apply on the next Step 1 fetch).
request_throttle_sec = 0.5      # pause between downloads; raise if Yahoo rate-limits you
min_rows_daily = 250            # trading days a stock needs or it's skipped (~1 year)
refresh_if_older_than_hours = 12   # re-use files downloaded within this many hours (resumes an interrupted run); 0 = always re-download
prune_stale = true              # after a full download, delete on-disk data for stocks no longer in your universe (keeps the disk tidy)

[output]
dir = output                    # folder where the report is written
